class date_clock_thread_update(QThread):
    time_signal = Signal(str,str,str)
    def __init__(self,web):
        super().__init__()
        self.web = web
    
    def run(self):
        #获取时间和日期
        while True:
            time = datetime.now().strftime('%H:%M:%S')
            date = datetime.now().strftime('%Y-%m-%d')
            position = self.web.url().toString()
            self.time_signal.emit(time,date,position)
            self.msleep(1)
    def stop(self):
        self.terminate()

    # 鼠标拖动frame 拖动窗口
    def mousePressEvent(self, event):
        #检查左键是否点到frame
        if self.ui.frame.geometry().contains(event.position().toPoint()) and event.button() == Qt.LeftButton:
            self.dragPosition = event.globalPosition().toPoint() - self.frameGeometry().topLeft()
            event.accept()
    def mouseMoveEvent(self, event):
        if self.dragPosition is not None and event.buttons() == Qt.LeftButton:
            self.move(event.globalPosition().toPoint() - self.dragPosition)
            event.accept()
    def mouseReleaseEvent(self, event):
        if event.button() == Qt.LeftButton:
            self.dragPosition = None
            event.accept()
    #双击 frame 放大或者缩小窗口
    def mouseDoubleClickEvent(self, event):
        self.change_window()
    #放大或者缩小窗口判断
    def change_window(self):
        if self.isMaximized():
            self.showNormal()
        else:
            self.showMaximized()
